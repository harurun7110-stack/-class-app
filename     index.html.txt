<script type="module">
import { initializeApp } from "https://www.gstatic.com/firebasejs/10.7.1/firebase-app.js";
import { getAuth, createUserWithEmailAndPassword, signInWithEmailAndPassword }
from "https://www.gstatic.com/firebasejs/10.7.1/firebase-auth.js";
import { getFirestore, doc, setDoc, getDoc, updateDoc }
from "https://www.gstatic.com/firebasejs/10.7.1/firebase-firestore.js";

const firebaseConfig = {
  apiKey: "AIzaSyBxxDI8P9JS69swA5hI4sxioyBeW2c6nQA",
  authDomain: "nen3zyouhouy.firebaseapp.com",
  projectId: "nen3zyouhouy"
};

const app = initializeApp(firebaseConfig);
const auth = getAuth();
const db = getFirestore();

let uid = "";

// 登録
window.register = async () => {
  const name = nameInput.value;
  const email = emailInput.value;
  const pass = passwordInput.value;

  const user = await createUserWithEmailAndPassword(auth, email, pass);
  uid = user.user.uid;

  await setDoc(doc(db,"users",uid),{
    name:name,
    point:100
  });

  msg.innerText="登録成功";
}

// ログイン
window.loginUser = async () => {
  const email = emailInput.value;
  const pass = passwordInput.value;

  const user = await signInWithEmailAndPassword(auth, email, pass);
  uid = user.user.uid;

  loadUser();
}

// ユーザー読み込み
async function loadUser(){
  const snap = await getDoc(doc(db,"users",uid));
  const data = snap.data();

  login.style.display="none";
  home.style.display="block";

  username.innerText = data.name;
  point.innerText = "ポイント："+data.point;
}

// ショップ
window.openShop = () => {
  content.innerHTML = `
  <h3>ショップ</h3>
  <button onclick="buyItem()">情報（10pt）</button>
  `;
}

// 購入
window.buyItem = async () => {
  const ref = doc(db,"users",uid);
  const snap = await getDoc(ref);
  let p = snap.data().point;

  if(p >= 10){
    if(confirm("購入しますか？")){
      await updateDoc(ref,{point:p-10});
      alert("購入ありがとう！");
      loadUser();
    }
  } else {
    alert("ポイント足りない");
  }
}

// ポイント表示
window.showPoint = async () => {
  const snap = await getDoc(doc(db,"users",uid));
  alert("ポイント："+snap.data().point);
}

// 管理者
window.openAdmin = () => {
  const code = prompt("コード入力");
  if(code === "582854"){
    content.innerHTML = `
    <h3>管理者</h3>
    <button onclick="addPoint()">+10ポイント</button>
    `;
  } else {
    alert("違う");
  }
}

// ポイント追加
window.addPoint = async () => {
  const ref = doc(db,"users",uid);
  const snap = await getDoc(ref);
  let p = snap.data().point;

  await updateDoc(ref,{point:p+10});
  loadUser();
}
